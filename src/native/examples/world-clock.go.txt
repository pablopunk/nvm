package main

import (
	"fmt"
	"strings"
	"time"

	"nvm"
)

var cities = []struct{ Name, Zone string }{
	{"Tokyo", "Asia/Tokyo"}, {"London", "Europe/London"}, {"New York", "America/New_York"},
}

func rows() []nvm.Item {
	now := time.Now()
	var items []nvm.Item
	for _, c := range cities {
		loc, err := time.LoadLocation(c.Zone)
		if err != nil {
			continue
		}
		t := now.In(loc).Format("15:04:05")
		items = append(items, nvm.Item{Title: fmt.Sprintf("%s — %s", c.Name, t), Subtitle: "World Clock · " + c.Zone, Value: t})
	}
	return items
}

func Extension() nvm.Extension {
	return nvm.Extension{
		ID:    "poc.world-clock",
		Title: "World Clock",
		Search: func(q string) []nvm.Item {
			if !strings.Contains("world clock time", strings.ToLower(q)) || len(q) < 3 {
				return nil
			}
			return rows()
		},
		MenuBar: func() (string, []nvm.MenuRow) {
			var r []nvm.MenuRow
			for _, it := range rows() {
				r = append(r, nvm.MenuRow{Title: it.Title, Value: it.Value})
			}
			return time.Now().UTC().Format("15:04"), r
		},
	}
}
